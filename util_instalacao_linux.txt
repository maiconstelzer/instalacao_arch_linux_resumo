# Instalacao
loadkeys us
setfont ter-112n
timedatectl set-timezone America/Sao_Paulo
timedatectl set-ntp true
mount /dev/sda2 /mnt
pacstrap -K /mnt base linux linux-firmware nano grub
genfstab -U /mnt >> /mnt/etc/fstab
arch-chroot -S /mnt
ln -sf /usr/share/zoneinfo/America/Sao_Paulo /etc/localtime
hwclock --systohc
locale-gen (nano /etc/locale.gen descomentar en_US.UTF-8)
nano /etc/locale.conf (LANG=en_US.UTF-8)
nano /etc/hostname (rat)
passwd

# GRUB
grub-install --target=i386-pc /dev/sda
grub-mkconfig -o /boot/grub/grub.cfg

# Particionamento BIOS/MBR
fdisk /dev/sda -> o (nova particao mbr) -> w (gravar)
cfdisk /dev/sda -> /dev/sda1 4GB (primary)
		-> /dev/sda2 RESTO (primary) (bootavel)
/dev/sda1 -> 82: Linux Swap -> mkswap /dev/sda1 (swapon /dev/sda1)
/dev/sda2 -> 80: Linux -> mkfs.ext4 /dev/sda2

# Console (instalar terminus-font) (/etc/vconsole.conf)
KEYMAP=us
FONT=ter-112n
FONT_MAP=8859-2

# Usuario
useradd -m -G wheel mds
passwd mds

# Programas 
pacman -S i3-wm i3status mpv ffmpeg xterm xorg-xinit firefox dosfstools 7zip ranger w3m feh scrot xorg-setxkbmap xorg-server mesa vulkan-radeon nginx php-fpm unzip dmenu xclip retroarch libretro-mupen64plus-next yt-dlp terminus-font libretro-mgba libretro-bsnes android-tools libretro-ppsspp util-linux usbutils git openssh
(ttf-dejavu jack2)

# Xinit (.xinitrc)
setxkbmap -layout us -variant intl &
i3
